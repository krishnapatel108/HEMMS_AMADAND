// ═══════════════════════════════════════════════════════════════
//  ENTRY POINTS
// ═══════════════════════════════════════════════════════════════

function doPost(e) {
  try {
    var p = JSON.parse(e.postData.contents);
    switch (p.mode || "append") {
      case "append":        return j(handleAppend(p.report));
      case "upsert":        return j(handleUpsert(p.reports));
      case "update_status": return j(handleUpdateStatus(p));
      default:              return j({ ok: false, error: "Unknown mode" });
    }
  } catch (err) { return j({ ok: false, error: err.message }); }
}

function doGet() {
  return j({ ok: true, msg: "HEMM v2.2 — 20 features", features: 20 });
}

// ═══════════════════════════════════════════════════════════════
//  HANDLERS
// ═══════════════════════════════════════════════════════════════

function handleAppend(r) {
  if (!r) return { ok: false, error: "No report" };
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var masterSh = getSheet(ss, MASTER_SHEET);

  // Dedup check: skip if this report was already appended (prevents duplicates from retries)
  if (r.firebaseKey && findRow(masterSh, r.firebaseKey) > 0) {
    return { ok: true, mode: "append", id: r.firebaseKey, skipped: true };
  }

  addRow(masterSh, r);
  addRow(getSheet(ss, dailyName(r.time)), r);
  return { ok: true, mode: "append", id: r.firebaseKey || "" };
}

function handleUpsert(reports) {
  if (!reports || !reports.length) return { ok: false, error: "No reports" };
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = getSheet(ss, MASTER_SHEET);

  // Batch-load all notes at once (O(1) instead of O(n) per report)
  var noteMap = buildNoteMap(sh);

  var added = 0, updated = 0;
  reports.forEach(function(r) {
    var row = noteMap[r.firebaseKey] || -1;
    if (row > 0) {
      sh.getRange(row, 1, 1, TC).setValues([buildRow(r)]);
      sh.getRange(row, 1).setNote(r.firebaseKey || "");
      styleRow(sh, row, r);
      updated++;
    } else {
      addRow(sh, r);
      // Update noteMap so subsequent reports with same key won't duplicate
      if (r.firebaseKey) noteMap[r.firebaseKey] = sh.getLastRow();
      added++;
    }
  });
  return { ok: true, mode: "upsert", added: added, updated: updated, count: added + updated };
}

function handleUpdateStatus(p) {
  if (!p.firebaseKey || !p.status) return { ok: false, error: "Missing data" };
  var sh = getSheet(SpreadsheetApp.getActiveSpreadsheet(), MASTER_SHEET);
  var row = findRow(sh, p.firebaseKey);
  if (row <= 0) return { ok: false, error: "Not found" };
  var c = sh.getRange(row, TC);
  c.setValue(p.status.toUpperCase());
  styleStatus(c, p.status);
  return { ok: true, mode: "update_status", id: p.firebaseKey };
}

// ═══════════════════════════════════════════════════════════════
//  SHEET CREATION — Clean Professional Headers
// ═══════════════════════════════════════════════════════════════

function getSheet(ss, name) {
  var sh = ss.getSheetByName(name);
  if (sh) return sh;
  sh = ss.insertSheet(name);

  // ── ROW 1: Group header row ──
  // Fill entire row dark
  var r1 = sh.getRange(1, 1, 1, TC);
  r1.setBackground("#1A1A2E");

  // Report Info (cols 1-4)
  var info = sh.getRange(1, 1, 1, 4);
  info.merge().setValue("📋 REPORT INFO");
  info.setBackground("#1A1A2E").setFontColor("#FFFFFF")
    .setFontWeight("bold").setFontSize(9)
    .setHorizontalAlignment("center").setVerticalAlignment("middle");

  // Feature groups
  FEATURE_GROUPS.forEach(function(g) {
    var col = FC + g.start;
    var span = g.end - g.start + 1;
    var rng = sh.getRange(1, col, 1, span);
    rng.merge().setValue(g.name);
    rng.setBackground(g.color).setFontColor("#FFFFFF")
      .setFontWeight("bold").setFontSize(8)
      .setHorizontalAlignment("center").setVerticalAlignment("middle");
  });

  // Operator Info (last 4 cols)
  var op = sh.getRange(1, LC + 1, 1, 4);
  op.merge().setValue("👷 OPERATOR INFO");
  op.setBackground("#263238").setFontColor("#FFFFFF")
    .setFontWeight("bold").setFontSize(9)
    .setHorizontalAlignment("center").setVerticalAlignment("middle");

  // ── ROW 2: Column sub-headers ──
  var hdrs = ["Machine", "Door #", "Reported At", "Note"]
    .concat(DGMS_HEADERS)
    .concat(["Operator", "Form A", "Auth", "Status"]);
  sh.getRange(2, 1, 1, TC).setValues([hdrs]);

  var r2 = sh.getRange(2, 1, 1, TC);
  r2.setBackground("#2C3E50").setFontColor("#ECF0F1")
    .setFontWeight("bold").setFontSize(8)
    .setHorizontalAlignment("center").setVerticalAlignment("middle")
    .setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);

  // ── Sizing ──
  sh.setRowHeight(1, 30);
  sh.setRowHeight(2, 38);
  sh.setColumnWidth(1, 85);   // Machine
  sh.setColumnWidth(2, 65);   // Door #
  sh.setColumnWidth(3, 135);  // Reported At
  sh.setColumnWidth(4, 160);  // Note
  for (var i = FC; i <= LC; i++) sh.setColumnWidth(i, 70);
  sh.setColumnWidth(LC + 1, 115); // Operator
  sh.setColumnWidth(LC + 2, 75);  // Form A
  sh.setColumnWidth(LC + 3, 95);  // Auth
  sh.setColumnWidth(LC + 4, 85);  // Status

  // ── Freeze ──
  sh.setFrozenRows(2);
  sh.setFrozenColumns(4);

  // ── Thin borders on headers ──
  sh.getRange(1, 1, 2, TC).setBorder(
    true, true, true, true, true, true,
    "#7F8C8D", SpreadsheetApp.BorderStyle.SOLID);

  return sh;
}

// ═══════════════════════════════════════════════════════════════
//  ROW DATA & FORMATTING
// ═══════════════════════════════════════════════════════════════

function buildRow(r) {
  var issues = toSet(r.problems || "");
  var row = [
    r.machine || "—",
    r.tipperNo || "—",
    fmtTime(r.time),
    r.note || ""
  ];
  DGMS_FEATURES.forEach(function(f) {
    row.push(issues.has(f) ? "✘" : "✔");
  });
  row.push(r.operatorName || "—");
  row.push(r.operatorFormA || "—");
  row.push((r.operatorAuth || "unknown").toUpperCase());
  row.push((r.status || "pending").toUpperCase());
  return row;
}

function addRow(sh, r) {
  var vals = buildRow(r);
  sh.appendRow(vals);
  var row = sh.getLastRow();
  if (r.firebaseKey) sh.getRange(row, 1).setNote(r.firebaseKey);
  styleRow(sh, row, r);
}

function styleRow(sh, row, r) {
  var full = sh.getRange(row, 1, 1, TC);

  // ── Alternating row color ──
  var bg = (row % 2 === 0) ? ROW_GREY : ROW_WHITE;
  full.setBackground(bg);
  full.setFontSize(9).setFontColor("#2C3E50");  // dark text on light bg
  full.setVerticalAlignment("middle").setHorizontalAlignment("center");
  sh.setRowHeight(row, 28);

  // ── Text columns left-aligned ──
  sh.getRange(row, 1).setHorizontalAlignment("left").setFontWeight("bold");
  sh.getRange(row, 2).setFontWeight("bold").setFontSize(12);
  sh.getRange(row, 4).setHorizontalAlignment("left").setFontSize(8).setFontColor("#5D6D7E");
  sh.getRange(row, LC + 1).setHorizontalAlignment("left");

  // ── Feature cells: ✔ green / ✘ red with clear contrast ──
  for (var i = 0; i < DGMS_FEATURES.length; i++) {
    var cell = sh.getRange(row, FC + i);
    var v = String(cell.getValue());
    if (v === "✘") {
      cell.setFontColor("#FFFFFF").setBackground("#E74C3C").setFontWeight("bold").setFontSize(10);
    } else {
      cell.setFontColor("#27AE60").setBackground(bg).setFontSize(10);
    }
  }

  // ── Auth cell — colored text only, no bg color ──
  var authCell = sh.getRange(row, LC + 3);
  authCell.setBackground(bg);  // keep row bg
  var auth = String(authCell.getValue()).toLowerCase();
  if (auth === "authorized") {
    authCell.setFontColor("#1B8C3A").setFontWeight("bold").setFontSize(8);
  } else if (auth === "unauthorized") {
    authCell.setFontColor("#C0392B").setFontWeight("bold").setFontSize(8);
  } else {
    authCell.setFontColor("#95A5A6").setFontSize(8);
  }

  // ── Status cell — colored text only, no bg color ──
  styleStatus(sh.getRange(row, TC), r.status || "pending", bg);

  // ── Thin bottom border ──
  full.setBorder(null, null, true, null, null, null,
    "#D5D8DC", SpreadsheetApp.BorderStyle.SOLID);
}

function styleStatus(cell, status, bg) {
  var s = String(status).toLowerCase();
  cell.setValue(s.toUpperCase()).setFontWeight("bold").setFontSize(8);
  cell.setBackground(bg || "#FFFFFF");  // keep row bg
  if (s === "pending") {
    cell.setFontColor("#D35400");
  } else if (s === "reviewing") {
    cell.setFontColor("#2471A3");
  } else if (s === "resolved") {
    cell.setFontColor("#1B8C3A");
  } else {
    cell.setFontColor("#95A5A6");
  }
}

// ═══════════════════════════════════════════════════════════════
//  UTILITIES
// ═══════════════════════════════════════════════════════════════

function toSet(p) {
  var arr = Array.isArray(p) ? p : p.split(",");
  return new Set(arr.map(function(s) { return s.trim(); }).filter(Boolean));
}

function findRow(sh, key) {
  if (!key) return -1;
  var last = sh.getLastRow();
  if (last < 3) return -1;
  var notes = sh.getRange(3, 1, last - 2, 1).getNotes();
  for (var i = 0; i < notes.length; i++) {
    if (notes[i][0] === key) return i + 3;
  }
  return -1;
}

// Batch-load all firebaseKey notes into a { key: rowNumber } map
// Used by handleUpsert to avoid calling findRow per-report (O(n^2) -> O(n))
function buildNoteMap(sh) {
  var map = {};
  var last = sh.getLastRow();
  if (last < 3) return map;
  var notes = sh.getRange(3, 1, last - 2, 1).getNotes();
  for (var i = 0; i < notes.length; i++) {
    var k = notes[i][0];
    if (k) map[k] = i + 3;
  }
  return map;
}

function fmtTime(iso) {
  try {
    var d = new Date(iso || Date.now());
    return d.toLocaleString("en-IN", {
      timeZone: "Asia/Kolkata",
      day: "2-digit", month: "short", year: "numeric",
      hour: "2-digit", minute: "2-digit"
    });
  } catch (_) { return iso || "—"; }
}

function dailyName(iso) {
  try {
    var d = new Date(iso || Date.now());
    var m = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
    return String(d.getDate()).padStart(2,"0") + "-" + m[d.getMonth()] + "-" + d.getFullYear();
  } catch (_) { return "Unknown"; }
}

function j(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
