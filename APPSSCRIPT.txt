// ══════════════════════════════════════════════════════════════
//  ENTRY POINT
// ══════════════════════════════════════════════════════════════
function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const mode    = payload.mode;

    if (mode === "append") {
      // Single report submitted by operator
      appendReport(payload);
      return jsonOk({ status: "ok" });

    } else if (mode === "upsert") {
      // Bulk export from Engineer/Admin dashboard
      const result = upsertReports(payload);
      return jsonOk({ status: "ok", added: result.added, updated: result.updated });

    } else if (mode === "update_status") {
      // Status changed on dashboard
      updateStatus(payload);
      return jsonOk({ status: "ok" });

    } else {
      return jsonOk({ status: "error", message: "Unknown mode: " + mode });
    }

  } catch (err) {
    return jsonOk({ status: "error", message: err.message });
  }
}

function doGet() {
  return ContentService
    .createTextOutput("HEMM AppScript v3.2 — OK")
    .setMimeType(ContentService.MimeType.TEXT);
}

function jsonOk(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ══════════════════════════════════════════════════════════════
//  APPEND — single new report (submitted by operator in the app)
// ══════════════════════════════════════════════════════════════
function appendReport(payload) {
  const ss  = SpreadsheetApp.getActiveSpreadsheet();
  const r   = payload.report;
  const key = payload.firebaseKey || "";

  const reportedDate = r.time ? new Date(r.time) : new Date();

  // problems is a comma-separated string of HINDI feature names
  const issueSet = buildIssueSet(r.problems || "");

  // Write to master sheet
  const masterSheet = getOrCreateSheet(ss, "HEMM Reports", true);
  writeRow(masterSheet, r, key, issueSet, reportedDate);

  // Write to daily sheet tab e.g. "10-May-2026"
  const daySheet = getOrCreateSheet(ss, formatDateAsSheetName(reportedDate), false);
  writeRow(daySheet, r, key, issueSet, reportedDate);
}

// ══════════════════════════════════════════════════════════════
//  UPSERT — bulk export from Engineer/Admin dashboard
//  Adds new rows, updates existing ones by firebaseKey.
//  Never deletes rows.
// ══════════════════════════════════════════════════════════════
function upsertReports(payload) {
  const ss      = SpreadsheetApp.getActiveSpreadsheet();
  const reports = payload.reports || [];
  let added = 0, updated = 0;

  const masterSheet = getOrCreateSheet(ss, "HEMM Reports", true);

  // Build a map of firebaseKey → row number from existing sheet notes
  const existingKeyMap = buildKeyMap(masterSheet);

  reports.forEach(r => {
    const key          = r.firebaseKey || "";
    const reportedDate = r.time ? new Date(r.time) : new Date();
    const issueSet     = buildIssueSet(r.problems || "");

    if (key && existingKeyMap[key]) {
      // Row exists — update it in place
      updateRowInPlace(masterSheet, existingKeyMap[key], r, issueSet, reportedDate);
      updated++;
    } else {
      // New row — append
      writeRow(masterSheet, r, key, issueSet, reportedDate);
      added++;
    }

    // Also write/update on the daily sheet for that report's date
    const dayName  = formatDateAsSheetName(reportedDate);
    const daySheet = getOrCreateSheet(ss, dayName, false);
    const dayKeyMap = buildKeyMap(daySheet);

    if (key && dayKeyMap[key]) {
      updateRowInPlace(daySheet, dayKeyMap[key], r, issueSet, reportedDate);
    } else {
      writeRow(daySheet, r, key, issueSet, reportedDate);
    }
  });

  return { added, updated };
}

// ══════════════════════════════════════════════════════════════
//  UPDATE STATUS — called when engineer changes status on dashboard
// ══════════════════════════════════════════════════════════════
function updateStatus(payload) {
  const ss          = SpreadsheetApp.getActiveSpreadsheet();
  const firebaseKey = payload.firebaseKey;
  const newStatus   = payload.status || "RESOLVED";
  const updatedAt   = payload.updatedAt || new Date().toLocaleString();

  ss.getSheets().forEach(sheet => {
    findAndUpdateStatus(sheet, firebaseKey, newStatus, updatedAt);
  });
}

// ══════════════════════════════════════════════════════════════
//  SHEET SETUP
// ══════════════════════════════════════════════════════════════
function getOrCreateSheet(ss, name, isMaster) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (isMaster) {
      ss.setActiveSheet(sheet);
      ss.moveActiveSheet(0);
    }
    buildHeader(sheet);
  } else if (sheet.getLastRow() === 0) {
    buildHeader(sheet);
  }
  return sheet;
}

function buildHeader(sheet) {
  // Use English headers for readability in the sheet
  const headers = [
    ...FIXED_FRONT_COLS,
    ...DGMS_HEADERS,      // English labels, but data matching uses Hindi names
    ...FIXED_BACK_COLS
  ];

  const headerRange = sheet.getRange(1, 1, 1, TOTAL_COLS);
  headerRange.setValues([headers]);

  headerRange
    .setBackground(COLOR_HEADER_BG)
    .setFontColor(COLOR_HEADER_FG)
    .setFontWeight("bold")
    .setFontSize(10)
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setWrap(true);

  sheet.setRowHeight(1, 52);
  sheet.setFrozenRows(1);
  sheet.setFrozenColumns(2);

  sheet.setColumnWidth(1, 95);
  sheet.setColumnWidth(2, 75);
  sheet.setColumnWidth(3, 140);
  sheet.setColumnWidth(4, 190);

  for (let c = FEAT_COL_START; c <= FEAT_COL_END; c++) {
    sheet.setColumnWidth(c, 120);
  }

  sheet.setColumnWidth(COL_OP_NAME, 145);
  sheet.setColumnWidth(COL_FORM_A,  85);
  sheet.setColumnWidth(COL_AUTH,    95);
  sheet.setColumnWidth(COL_STATUS,  90);

  headerRange.setBorder(
    null, null, true, null, null, null,
    "#7986cb", SpreadsheetApp.BorderStyle.MEDIUM
  );

  sheet.getRange(1, COL_OP_NAME).setBorder(
    null, true, null, null, null, null,
    "#7986cb", SpreadsheetApp.BorderStyle.MEDIUM
  );
}

// ══════════════════════════════════════════════════════════════
//  WRITE ONE ROW (append to bottom)
// ══════════════════════════════════════════════════════════════
function writeRow(sheet, r, firebaseKey, issueSet, reportedDate) {
  const lastRow = sheet.getLastRow();
  const dataRow = lastRow + 1;
  const isOdd   = (dataRow % 2 === 0);

  const timeStr = Utilities.formatDate(
    reportedDate, Session.getScriptTimeZone(), "dd-MMM-yyyy HH:mm"
  );

  const rowValues = buildRowValues(r, timeStr, issueSet);
  const rowRange  = sheet.getRange(dataRow, 1, 1, TOTAL_COLS);
  rowRange.setValues([rowValues]);

  applyRowStyles(sheet, dataRow, issueSet, r, isOdd);
  sheet.setRowHeight(dataRow, 30);
  sheet.getRange(dataRow, 1).setNote("firebaseKey:" + firebaseKey);
}

// ══════════════════════════════════════════════════════════════
//  UPDATE ROW IN PLACE (for upsert)
// ══════════════════════════════════════════════════════════════
function updateRowInPlace(sheet, dataRow, r, issueSet, reportedDate) {
  const isOdd   = (dataRow % 2 === 0);
  const timeStr = Utilities.formatDate(
    reportedDate, Session.getScriptTimeZone(), "dd-MMM-yyyy HH:mm"
  );

  const rowValues = buildRowValues(r, timeStr, issueSet);
  sheet.getRange(dataRow, 1, 1, TOTAL_COLS).setValues([rowValues]);
  applyRowStyles(sheet, dataRow, issueSet, r, isOdd);
  sheet.setRowHeight(dataRow, 30);
}

// ── Shared: build the flat array of values for one row ────────
function buildRowValues(r, timeStr, issueSet) {
  const rowValues = [
    r.machine  || "—",
    r.tipperNo || "—",
    timeStr,
    r.note     || "",
  ];

  DGMS_FEATURES.forEach(feat => {
    rowValues.push(issueSet.has(feat) ? "ISSUE" : "OK");
  });

  rowValues.push(
    r.operatorName  || "—",
    r.operatorFormA || "—",
    (r.operatorAuth || "unknown").toUpperCase().replace("NOT VERIFIED", "UNVERIFIED"),
    r.status ? r.status.toUpperCase() : "PENDING"
  );

  return rowValues;
}

// ── Shared: apply all cell styles for a data row ──────────────
function applyRowStyles(sheet, dataRow, issueSet, r, isOdd) {
  const isUnauth = (r.operatorAuth || "").toLowerCase().includes("unauth");
  const baseBg   = isUnauth ? COLOR_UNAUTH_ROW : (isOdd ? COLOR_ROW_ODD : COLOR_ROW_EVEN);

  sheet.getRange(dataRow, 1, 1, TOTAL_COLS)
    .setBackground(baseBg)
    .setVerticalAlignment("middle")
    .setFontSize(10)
    .setHorizontalAlignment("center");

  sheet.getRange(dataRow, 4).setHorizontalAlignment("left");
  sheet.getRange(dataRow, COL_OP_NAME).setHorizontalAlignment("left");

  // Feature cells
  DGMS_FEATURES.forEach((feat, idx) => {
    const cell = sheet.getRange(dataRow, FEAT_COL_START + idx);
    if (issueSet.has(feat)) {
      cell.setBackground(COLOR_ISSUE_BG).setFontColor(COLOR_ISSUE_FG).setFontWeight("bold");
    } else {
      cell.setBackground(isOdd ? COLOR_OK_BG_ODD : COLOR_OK_BG_EVEN)
          .setFontColor(COLOR_OK_FG).setFontWeight("normal");
    }
  });

  // Auth cell
  const authVal  = (r.operatorAuth || "").toLowerCase();
  const authCell = sheet.getRange(dataRow, COL_AUTH);
  if (authVal.includes("auth") && !authVal.includes("unauth")) {
    authCell.setBackground("#c8e6c9").setFontColor("#1b5e20").setFontWeight("bold");
  } else if (authVal.includes("unauth")) {
    authCell.setBackground("#ef9a9a").setFontColor("#b71c1c").setFontWeight("bold");
  } else {
    authCell.setFontColor("#757575");
  }

  // Status cell
  const statusVal  = (r.status || "pending").toUpperCase();
  const statusCell = sheet.getRange(dataRow, COL_STATUS);
  if (statusVal === "RESOLVED") {
    statusCell.setBackground("#c8e6c9").setFontColor("#1b5e20").setFontWeight("bold");
  } else if (statusVal === "REVIEWING") {
    statusCell.setBackground("#fff9c4").setFontColor("#f57f17").setFontWeight("bold");
  } else {
    statusCell.setBackground("#fffde7").setFontColor("#e65100").setFontWeight("bold");
  }

  // Separator border before operator columns
  sheet.getRange(dataRow, COL_OP_NAME).setBorder(
    null, true, null, null, null, null,
    "#90a4ae", SpreadsheetApp.BorderStyle.SOLID_MEDIUM
  );
}

// ══════════════════════════════════════════════════════════════
//  FIND ROW AND UPDATE STATUS ONLY
// ══════════════════════════════════════════════════════════════
function findAndUpdateStatus(sheet, firebaseKey, newStatus, updatedAt) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const notes = sheet.getRange(2, 1, lastRow - 1, 1).getNotes();
  for (let i = 0; i < notes.length; i++) {
    if (notes[i][0].includes("firebaseKey:" + firebaseKey)) {
      const dataRow    = i + 2;
      const statusCell = sheet.getRange(dataRow, COL_STATUS);
      statusCell.setValue(newStatus);

      if (newStatus === "RESOLVED") {
        statusCell.setBackground("#c8e6c9").setFontColor("#1b5e20").setFontWeight("bold");
      } else if (newStatus === "REVIEWING") {
        statusCell.setBackground("#fff9c4").setFontColor("#f57f17").setFontWeight("bold");
      } else {
        statusCell.setBackground("#fffde7").setFontColor("#e65100").setFontWeight("bold");
      }

      const noteCell     = sheet.getRange(dataRow, 4);
      const existingNote = noteCell.getValue() || "";
      const sep          = existingNote ? " | " : "";
      noteCell.setValue(existingNote + sep + "✓ " + newStatus + " @ " + updatedAt);

      break;
    }
  }
}

/**
 * Parses problems string into a Set of Hindi feature names.
 * Handles both comma-separated Hindi names AND English names.
 * Input: "रियर विजन सिस्टम, Proximity Warning Sensor"
 */
function buildIssueSet(problemsStr) {
  return new Set(
    problemsStr.split(",").map(p => p.trim()).filter(Boolean)
  );
}

/**
 * Scans all cell notes in col 1 and returns { firebaseKey: rowNumber }
 */
function buildKeyMap(sheet) {
  const lastRow = sheet.getLastRow();
  const map = {};
  if (lastRow < 2) return map;

  const notes = sheet.getRange(2, 1, lastRow - 1, 1).getNotes();
  notes.forEach((row, i) => {
    const note = row[0];
    if (note && note.startsWith("firebaseKey:")) {
      const key = note.replace("firebaseKey:", "").trim();
      if (key) map[key] = i + 2;  // 1-based row number
    }
  });
  return map;
}

function formatDateAsSheetName(date) {
  return Utilities.formatDate(date, Session.getScriptTimeZone(), "dd-MMM-yyyy");
}
