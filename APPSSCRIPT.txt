const TOTAL_COLS     = 4 + DGMS_FEATURES.length + 4; // 28

// ═══════════════════════════════════════════════════════════════
//  ENTRY POINT
// ═══════════════════════════════════════════════════════════════

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const mode = payload.mode || "append";
    switch (mode) {
      case "append":        return handleAppend(payload.report);
      case "upsert":        return handleUpsert(payload.reports);
      case "update_status": return handleUpdateStatus(payload);
      default:              return ok({ error: "Unknown mode: " + mode });
    }
  } catch (err) {
    return ok({ ok: false, error: err.message });
  }
}

function doGet() {
  return ok({
    ok: true,
    message: "HEMM Report Apps Script v2.1 Professional — 20 DGMS features",
    features: DGMS_FEATURES.length,
    columns: TOTAL_COLS,
  });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: APPEND
// ═══════════════════════════════════════════════════════════════

function handleAppend(report) {
  if (!report) return ok({ ok: false, error: "No report data" });
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const master = getOrCreateSheet(ss, MASTER_SHEET);
  appendAndFormat(master, report);

  const dailyName = getDailySheetName(report.time || new Date().toISOString());
  const daily = getOrCreateSheet(ss, dailyName);
  appendAndFormat(daily, report);

  return ok({ ok: true, mode: "append", id: report.firebaseKey || "" });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: UPSERT
// ═══════════════════════════════════════════════════════════════

function handleUpsert(reports) {
  if (!reports || !Array.isArray(reports))
    return ok({ ok: false, error: "No reports array" });

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = getOrCreateSheet(ss, MASTER_SHEET);
  let count = 0;

  reports.forEach(function (report) {
    const rowIdx = findRowByKey(master, report.firebaseKey);
    if (rowIdx > 0) {
      const vals = buildRowValues(report);
      master.getRange(rowIdx, 1, 1, vals.length).setValues([vals]);
      master.getRange(rowIdx, 1).setNote(report.firebaseKey || "");
      formatDataRow(master, rowIdx, report);
    } else {
      appendAndFormat(master, report);
    }
    count++;
  });

  return ok({ ok: true, mode: "upsert", count: count });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: UPDATE_STATUS
// ═══════════════════════════════════════════════════════════════

function handleUpdateStatus(payload) {
  const key = payload.firebaseKey, status = payload.status;
  if (!key || !status) return ok({ ok: false, error: "Missing firebaseKey or status" });

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = getOrCreateSheet(ss, MASTER_SHEET);
  const rowIdx = findRowByKey(master, key);
  if (rowIdx <= 0) return ok({ ok: false, error: "Report not found: " + key });

  const statusCol = TOTAL_COLS;
  master.getRange(rowIdx, statusCol).setValue(status.toUpperCase());
  formatStatusCell(master, rowIdx, statusCol, status);

  return ok({ ok: true, mode: "update_status", id: key });
}

// ═══════════════════════════════════════════════════════════════
//  SHEET CREATION & PROFESSIONAL FORMATTING
// ═══════════════════════════════════════════════════════════════

function getOrCreateSheet(ss, name) {
  let sheet = ss.getSheetByName(name);
  if (sheet) return sheet;

  sheet = ss.insertSheet(name);

  // ── ROW 1: Group Headers (merged cells spanning feature groups) ──
  const groupRow = new Array(TOTAL_COLS).fill("");
  groupRow[0] = "";  // empty above basic info cols
  sheet.getRange(1, 1, 1, TOTAL_COLS).setValues([groupRow]);

  // Merge & style basic info header (cols 1-4)
  const infoMerge = sheet.getRange(1, 1, 1, 4);
  infoMerge.merge();
  infoMerge.setValue("REPORT INFO");
  infoMerge.setBackground(COLORS.headerBg);
  infoMerge.setFontColor(COLORS.headerText);
  infoMerge.setFontWeight("bold");
  infoMerge.setFontSize(9);
  infoMerge.setHorizontalAlignment("center");
  infoMerge.setVerticalAlignment("middle");

  // Merge & style each feature group
  FEATURE_GROUPS.forEach(function (g) {
    const startCol = FEAT_START_COL + g.start;
    const span = g.end - g.start + 1;
    const range = sheet.getRange(1, startCol, 1, span);
    range.merge();
    range.setValue(g.name);
    range.setBackground(g.color);
    range.setFontColor("#FFFFFF");
    range.setFontWeight("bold");
    range.setFontSize(8);
    range.setHorizontalAlignment("center");
    range.setVerticalAlignment("middle");
  });

  // Merge & style operator info header (last 4 cols)
  const opCol = FEAT_END_COL + 1;
  const opMerge = sheet.getRange(1, opCol, 1, 4);
  opMerge.merge();
  opMerge.setValue("OPERATOR INFO");
  opMerge.setBackground("#263238");
  opMerge.setFontColor(COLORS.headerText);
  opMerge.setFontWeight("bold");
  opMerge.setFontSize(9);
  opMerge.setHorizontalAlignment("center");
  opMerge.setVerticalAlignment("middle");

  // ── ROW 2: Sub-headers (individual column names) ──
  const headers = buildHeaders();
  sheet.getRange(2, 1, 1, headers.length).setValues([headers]);

  const hdrRange = sheet.getRange(2, 1, 1, TOTAL_COLS);
  hdrRange.setBackground(COLORS.subHeaderBg);
  hdrRange.setFontColor(COLORS.subHeaderTxt);
  hdrRange.setFontWeight("bold");
  hdrRange.setFontSize(8);
  hdrRange.setHorizontalAlignment("center");
  hdrRange.setVerticalAlignment("middle");
  hdrRange.setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);

  // ── Row Heights ──
  sheet.setRowHeight(1, 28);
  sheet.setRowHeight(2, 36);

  // ── Column Widths ──
  sheet.setColumnWidth(1, 90);    // Machine
  sheet.setColumnWidth(2, 70);    // Door No.
  sheet.setColumnWidth(3, 140);   // Reported At
  sheet.setColumnWidth(4, 150);   // Note
  for (var i = FEAT_START_COL; i <= FEAT_END_COL; i++) {
    sheet.setColumnWidth(i, 72);  // Feature columns
  }
  sheet.setColumnWidth(FEAT_END_COL + 1, 120);  // Operator Name
  sheet.setColumnWidth(FEAT_END_COL + 2, 80);   // Form A
  sheet.setColumnWidth(FEAT_END_COL + 3, 90);   // Auth
  sheet.setColumnWidth(FEAT_END_COL + 4, 85);   // Status

  // ── Freeze ──
  sheet.setFrozenRows(2);
  sheet.setFrozenColumns(2);

  // ── Borders for header rows ──
  var allHeaders = sheet.getRange(1, 1, 2, TOTAL_COLS);
  allHeaders.setBorder(true, true, true, true, true, true,
    COLORS.borderColor, SpreadsheetApp.BorderStyle.SOLID);

  return sheet;
}

function buildHeaders() {
  var base = ["Machine", "Door #", "Reported At", "Note"];
  var features = DGMS_HEADERS.slice();
  var tail = ["Operator", "Form A", "Auth", "Status"];
  return base.concat(features).concat(tail);
}

// ═══════════════════════════════════════════════════════════════
//  ROW BUILDING & FORMATTING
// ═══════════════════════════════════════════════════════════════

function buildRowValues(report) {
  var issueSet = buildIssueSet(report.problems || "");
  var row = [];

  row.push(report.machine || "—");
  row.push(report.tipperNo || "—");
  row.push(formatTime(report.time));
  row.push(report.note || "");

  DGMS_FEATURES.forEach(function (feat) {
    row.push(issueSet.has(feat) ? "✘ ISSUE" : "✔ OK");
  });

  row.push(report.operatorName || "—");
  row.push(report.operatorFormA || "—");
  row.push((report.operatorAuth || "unknown").toUpperCase());
  row.push((report.status || "pending").toUpperCase());

  return row;
}

function appendAndFormat(sheet, report) {
  var vals = buildRowValues(report);
  sheet.appendRow(vals);
  var row = sheet.getLastRow();

  // Set firebaseKey as note
  if (report.firebaseKey) {
    sheet.getRange(row, 1).setNote(report.firebaseKey);
  }

  formatDataRow(sheet, row, report);
}

function formatDataRow(sheet, row, report) {
  var fullRow = sheet.getRange(row, 1, 1, TOTAL_COLS);

  // ── Alternating row background ──
  var isEven = (row % 2 === 0);
  fullRow.setBackground(isEven ? COLORS.rowEven : COLORS.rowOdd);

  // ── Font defaults ──
  fullRow.setFontSize(9);
  fullRow.setVerticalAlignment("middle");
  fullRow.setHorizontalAlignment("center");

  // ── Left-align text columns ──
  sheet.getRange(row, 1).setHorizontalAlignment("left");   // Machine
  sheet.getRange(row, 4).setHorizontalAlignment("left");   // Note
  sheet.getRange(row, FEAT_END_COL + 1).setHorizontalAlignment("left"); // Operator name

  // ── Bold machine & door ──
  sheet.getRange(row, 1).setFontWeight("bold");
  sheet.getRange(row, 2).setFontWeight("bold").setFontSize(11);

  // ── Format feature cells (OK / ISSUE) ──
  for (var i = 0; i < DGMS_FEATURES.length; i++) {
    var cell = sheet.getRange(row, FEAT_START_COL + i);
    var val = cell.getValue();
    if (String(val).indexOf("ISSUE") !== -1) {
      cell.setBackground(COLORS.issueBg);
      cell.setFontColor(COLORS.issueText);
      cell.setFontWeight("bold");
    } else {
      cell.setBackground(COLORS.okBg);
      cell.setFontColor(COLORS.okText);
    }
    cell.setFontSize(8);
  }

  // ── Format Auth cell ──
  var authCol = FEAT_END_COL + 3;
  var authCell = sheet.getRange(row, authCol);
  var authVal = String(authCell.getValue()).toLowerCase();
  if (authVal === "authorized") {
    authCell.setBackground(COLORS.authOkBg);
    authCell.setFontColor("#1B5E20");
    authCell.setFontWeight("bold");
  } else if (authVal === "unauthorized") {
    authCell.setBackground(COLORS.authNoBg);
    authCell.setFontColor("#B71C1C");
    authCell.setFontWeight("bold");
  } else {
    authCell.setBackground(COLORS.authUnkBg);
    authCell.setFontColor("#F57F17");
  }

  // ── Format Status cell ──
  formatStatusCell(sheet, row, TOTAL_COLS, report.status || "pending");

  // ── Row border ──
  fullRow.setBorder(null, null, true, null, null, null,
    "#E0E0E0", SpreadsheetApp.BorderStyle.SOLID);

  // ── Row height ──
  sheet.setRowHeight(row, 30);
}

function formatStatusCell(sheet, row, col, status) {
  var cell = sheet.getRange(row, col);
  var s = String(status).toLowerCase();
  cell.setValue(s.toUpperCase());
  cell.setFontWeight("bold");
  cell.setFontSize(8);
  if (s === "pending") {
    cell.setBackground(COLORS.statusPending);
    cell.setFontColor("#E65100");
  } else if (s === "reviewing") {
    cell.setBackground(COLORS.statusReview);
    cell.setFontColor("#0D47A1");
  } else if (s === "resolved") {
    cell.setBackground(COLORS.statusResolved);
    cell.setFontColor("#1B5E20");
  }
}

// ═══════════════════════════════════════════════════════════════
//  HELPERS
// ═══════════════════════════════════════════════════════════════

function buildIssueSet(problemsStr) {
  if (Array.isArray(problemsStr)) {
    return new Set(problemsStr.map(function (p) { return p.trim(); }).filter(Boolean));
  }
  return new Set(
    problemsStr.split(",").map(function (p) { return p.trim(); }).filter(Boolean)
  );
}

function findRowByKey(sheet, key) {
  if (!key) return -1;
  var lastRow = sheet.getLastRow();
  if (lastRow < 3) return -1;  // rows 1-2 are headers
  var notes = sheet.getRange(3, 1, lastRow - 2, 1).getNotes();
  for (var i = 0; i < notes.length; i++) {
    if (notes[i][0] === key) return i + 3;
  }
  return -1;
}

function formatTime(isoStr) {
  try {
    if (!isoStr) return new Date().toLocaleString("en-IN");
    var d = new Date(isoStr);
    return d.toLocaleString("en-IN", {
      timeZone: "Asia/Kolkata",
      day: "2-digit", month: "short", year: "numeric",
      hour: "2-digit", minute: "2-digit",
    });
  } catch (_) {
    return isoStr || "—";
  }
}

function getDailySheetName(isoStr) {
  try {
    var d = new Date(isoStr);
    var day = String(d.getDate()).padStart(2, "0");
    var months = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
    return day + "-" + months[d.getMonth()] + "-" + d.getFullYear();
  } catch (_) {
    return "Unknown-Date";
  }
}

function ok(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
