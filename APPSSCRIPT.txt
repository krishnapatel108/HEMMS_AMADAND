// ═══════════════════════════════════════════════════════════════
//  ENTRY POINT — handles POST requests from the web app
// ═══════════════════════════════════════════════════════════════

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const mode = payload.mode || "append";

    switch (mode) {
      case "append":
        return handleAppend(payload.report);
      case "upsert":
        return handleUpsert(payload.reports);
      case "update_status":
        return handleUpdateStatus(payload);
      default:
        return jsonResponse({ ok: false, error: "Unknown mode: " + mode });
    }
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message });
  }
}

// Allow GET for testing
function doGet(e) {
  return jsonResponse({
    ok: true,
    message: "HEMM Report Apps Script v2.0 — 20 DGMS features",
    features: DGMS_FEATURES.length,
  });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: APPEND — Single report from operator
// ═══════════════════════════════════════════════════════════════

function handleAppend(report) {
  if (!report) return jsonResponse({ ok: false, error: "No report data" });

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // Write to master sheet
  const master = getOrCreateSheet(ss, MASTER_SHEET);
  appendReportRow(master, report);

  // Write to daily sheet
  const dailyName = getDailySheetName(report.time || new Date().toISOString());
  const daily = getOrCreateSheet(ss, dailyName);
  appendReportRow(daily, report);

  return jsonResponse({ ok: true, mode: "append", id: report.firebaseKey || "" });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: UPSERT — Bulk export from dashboard
// ═══════════════════════════════════════════════════════════════

function handleUpsert(reports) {
  if (!reports || !Array.isArray(reports)) {
    return jsonResponse({ ok: false, error: "No reports array" });
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = getOrCreateSheet(ss, MASTER_SHEET);
  let upserted = 0;

  reports.forEach(function (report) {
    const rowIdx = findRowByKey(master, report.firebaseKey);
    if (rowIdx > 0) {
      // Update existing row
      const rowValues = buildRowValues(report);
      master.getRange(rowIdx, 1, 1, rowValues.length).setValues([rowValues]);
      // Re-add the key note
      master.getRange(rowIdx, 1).setNote(report.firebaseKey || "");
    } else {
      // Append new row
      appendReportRow(master, report);
    }
    upserted++;
  });

  return jsonResponse({ ok: true, mode: "upsert", count: upserted });
}

// ═══════════════════════════════════════════════════════════════
//  MODE: UPDATE_STATUS — Status change from dashboard
// ═══════════════════════════════════════════════════════════════

function handleUpdateStatus(payload) {
  const key = payload.firebaseKey;
  const status = payload.status;
  if (!key || !status) {
    return jsonResponse({ ok: false, error: "Missing firebaseKey or status" });
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const master = getOrCreateSheet(ss, MASTER_SHEET);
  const rowIdx = findRowByKey(master, key);

  if (rowIdx <= 0) {
    return jsonResponse({ ok: false, error: "Report not found: " + key });
  }

  // Status is the last column (col 28)
  const statusCol = 4 + DGMS_FEATURES.length + 4; // 28
  master.getRange(rowIdx, statusCol).setValue(status.toUpperCase());

  return jsonResponse({ ok: true, mode: "update_status", id: key });
}

// ═══════════════════════════════════════════════════════════════
//  HELPERS — Sheet Management
// ═══════════════════════════════════════════════════════════════

function getOrCreateSheet(ss, name) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    // Write header row
    const headers = buildHeaders();
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

    // Format header row
    const headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setFontWeight("bold");
    headerRange.setBackground("#1a1a2e");
    headerRange.setFontColor("#f5a623");
    headerRange.setHorizontalAlignment("center");
    headerRange.setWrap(true);

    // Freeze header row
    sheet.setFrozenRows(1);

    // Auto-resize first 4 columns
    for (let i = 1; i <= 4; i++) {
      sheet.setColumnWidth(i, 120);
    }
    // Feature columns (narrower)
    for (let i = 5; i <= 4 + DGMS_FEATURES.length; i++) {
      sheet.setColumnWidth(i, 80);
    }
    // Last 4 columns
    for (let i = 4 + DGMS_FEATURES.length + 1; i <= 4 + DGMS_FEATURES.length + 4; i++) {
      sheet.setColumnWidth(i, 120);
    }
  }
  return sheet;
}

function buildHeaders() {
  // Cols 1-4: Machine, Door No., Reported At, Note
  const base = ["Machine", "Door No.", "Reported At", "Note"];
  // Cols 5-24: 20 DGMS feature columns (English headers)
  const features = DGMS_HEADERS.slice();
  // Cols 25-28: Operator Name, Form A #, Auth, Status
  const tail = ["Operator Name", "Form A #", "Auth", "Status"];
  return base.concat(features).concat(tail);
}

// ═══════════════════════════════════════════════════════════════
//  HELPERS — Row Building
// ═══════════════════════════════════════════════════════════════

function buildRowValues(report) {
  const issueSet = buildIssueSet(report.problems || "");
  const rowValues = [];

  // Cols 1-4
  rowValues.push(report.machine || "—");
  rowValues.push(report.tipperNo || "—");
  rowValues.push(formatTime(report.time));
  rowValues.push(report.note || "");

  // Cols 5-24: DGMS features (OK or ISSUE)
  DGMS_FEATURES.forEach(function (feat) {
    rowValues.push(issueSet.has(feat) ? "ISSUE" : "OK");
  });

  // Cols 25-28
  rowValues.push(report.operatorName || "—");
  rowValues.push(report.operatorFormA || "—");
  rowValues.push((report.operatorAuth || "unknown").toUpperCase());
  rowValues.push((report.status || "pending").toUpperCase());

  return rowValues;
}

function buildIssueSet(problemsStr) {
  if (Array.isArray(problemsStr)) {
    return new Set(problemsStr.map(function (p) { return p.trim(); }).filter(Boolean));
  }
  return new Set(
    problemsStr.split(",").map(function (p) { return p.trim(); }).filter(Boolean)
  );
}

function appendReportRow(sheet, report) {
  const rowValues = buildRowValues(report);
  sheet.appendRow(rowValues);

  // Set a note on cell A with the firebaseKey for future lookup
  const lastRow = sheet.getLastRow();
  if (report.firebaseKey) {
    sheet.getRange(lastRow, 1).setNote(report.firebaseKey);
  }

  // Color-code the row based on auth status
  const auth = (report.operatorAuth || "unknown").toLowerCase();
  const rowRange = sheet.getRange(lastRow, 1, 1, rowValues.length);
  if (auth === "unauthorized") {
    rowRange.setBackground("#3d1515"); // Dark red tint
  } else if (auth === "authorized") {
    rowRange.setBackground("#153d15"); // Dark green tint
  }

  // Color-code ISSUE cells in red, OK cells in green
  for (let i = 0; i < DGMS_FEATURES.length; i++) {
    const cell = sheet.getRange(lastRow, 5 + i);
    if (cell.getValue() === "ISSUE") {
      cell.setFontColor("#ff4444");
      cell.setFontWeight("bold");
    } else {
      cell.setFontColor("#44ff44");
    }
  }
}

// ═══════════════════════════════════════════════════════════════
//  HELPERS — Row Lookup
// ═══════════════════════════════════════════════════════════════

function findRowByKey(sheet, key) {
  if (!key) return -1;
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;

  // Check notes on column A for matching firebaseKey
  const notes = sheet.getRange(2, 1, lastRow - 1, 1).getNotes();
  for (let i = 0; i < notes.length; i++) {
    if (notes[i][0] === key) {
      return i + 2; // +2 because row 1 is header and array is 0-indexed
    }
  }
  return -1;
}

// ═══════════════════════════════════════════════════════════════
//  HELPERS — Utilities
// ═══════════════════════════════════════════════════════════════

function formatTime(isoStr) {
  try {
    if (!isoStr) return new Date().toLocaleString("en-IN");
    const d = new Date(isoStr);
    return d.toLocaleString("en-IN", {
      timeZone: "Asia/Kolkata",
      year: "numeric",
      month: "short",
      day: "2-digit",
      hour: "2-digit",
      minute: "2-digit",
    });
  } catch (_) {
    return isoStr || "—";
  }
}

function getDailySheetName(isoStr) {
  try {
    const d = new Date(isoStr);
    const day = String(d.getDate()).padStart(2, "0");
    const months = [
      "Jan", "Feb", "Mar", "Apr", "May", "Jun",
      "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
    ];
    const month = months[d.getMonth()];
    const year = d.getFullYear();
    return day + "-" + month + "-" + year;
  } catch (_) {
    return "Unknown-Date";
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(
    ContentService.MimeType.JSON
  );
}
